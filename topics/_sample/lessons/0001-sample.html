Python Illustrative — written for this lesson 
from dataclasses import dataclass


@dataclass(frozen=True)
class OutboxRecord:
    """One pending message, written in the same transaction as the state change it describes."""

    aggregate_id: str
    payload: dict[str, str]
    published: bool = False

    def mark_published(self) -> "OutboxRecord":
        return OutboxRecord(self.aggregate_id, self.payload, published=True)
